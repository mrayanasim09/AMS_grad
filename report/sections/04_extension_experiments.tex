\section{Extension: Phase Boundary \& Mechanism Discrimination}
\label{sec:extension}

\subsection{A Priori Scaling Hypothesis}
We formulated the following scaling hypothesis for the divergence boundary of \textsc{Adam}:
\begin{quote}
\textbf{A Priori Scaling Hypothesis:} The divergence threshold $(1 - \beta_2^*)$ scales inversely with burst spacing $T_{\text{burst}} \approx \frac{C+1}{1+\delta}$, yielding a log-log slope of $\frac{\partial \log(1 - \beta_2^*)}{\partial \log C} \approx -1.00$.
\end{quote}
\paragraph{Decision Rule.} A fitted slope in $[-1.2, -0.8]$ supports the hypothesis; values outside refute it.

\subsection{Measurement Validity \& Stochastic Failure Surface}
Single-point terminal position $x_T$ is phase-dependent in periodic objectives: at $C=100$, $x_t$ oscillates across the full interval $[-1, 1]$ within each cycle. We therefore evaluate time-averaged metrics over the final full cycle:
\begin{equation}
    \bar{x} = \frac{1}{C} \sum_{t=T-C+1}^T x_t, \qquad \text{Dwell} = \frac{1}{C} \sum_{t=T-C+1}^T \mathbb{I}(x_t > 0.5).
    \label{eq:cycle_metric}
\end{equation}

\paragraph{Refutation of Simple Scaling.}
Rather than a universal $C^{-1}$ power law, the divergence boundary exists only for $C \le 70$, and disappears for $C \ge 75$ under fixed $\alpha=0.80$. As explained below, this disappearance is directly predicted by the $(\alpha, \beta_2, C)$ phase structure.

\subsection{Derivations \& Three-Regime Phase Structure}

\paragraph{Regime I: $\tau_2 \lesssim T_{\text{burst}}$ (v-Collapse).}
When $\beta_2 = 0.9$ ($\tau_2 = 10$), $v_t$ decays from $v_{\text{burst}} \approx C^2$ to $\approx 1$ in roughly $\tau_2$ steps. \textsc{Adam} steps against the gradient during the $C - 1$ non-burst steps at an amplified rate, failing across all tested learning rates $\alpha$.

\paragraph{Regime II: $\tau_2 \gg T_{\text{burst}}$ (Post-Burst Drift).}
When $\beta_2 = 0.999$ ($\tau_2 = 1000$), $\tilde{v}_t \approx \mathbb{E}[g^2] \approx C$ remains approximately flat throughout the cycle. We formally derive the analytical trajectory response:

\begin{enumerate}
    \item \textbf{Derivation of Protected Zero-Crossing $s^*$:}
    Following a burst $g_1 = +C$, the unrolled first moment is $\tilde{m}_t = \beta_1^t (1-\beta_1)C - (1 - \beta_1^t)$ for non-burst steps ($g_t = -1$). Setting $\tilde{m}_t = 0$ yields:
    \begin{equation}
        \beta_1^{s^*} \bigl((1-\beta_1)C + 1\bigr) = 1 \implies s^* = \frac{\ln\bigl((1-\beta_1)C + 1\bigr)}{\ln(1/\beta_1)}.
    \end{equation}
    Using the first-order expansion $\ln(1/\beta_1) = -\ln(1 - (1-\beta_1)) \approx 1 - \beta_1 = 1/\tau_1$, we obtain:
    \begin{equation}
        s^*(C) \approx \tau_1 \ln\bigl((1-\beta_1)C + 1\bigr),
        \label{eq:s_star_derived}
    \end{equation}
    yielding $s^*(100) \approx 23.98$ steps for $\beta_1=0.9$ ($\tau_1=10$).
    
    \item \textbf{Derivation of Traverse Condition $\alpha^*(C)$:}
    During steps $1 \to s^*$, $\tilde{m}_t > 0$ pushes leftward (retraction). For steps $s^*+1 \to C$, $\tilde{m}_t < 0$, so \textsc{Adam} steps rightward at rate $\frac{\alpha}{\sqrt{\tilde{v}_t}} \approx \frac{\alpha}{\sqrt{C}}$. Total rightward displacement over the remaining $C - s^*$ steps is $\Delta x_{\text{right}} = (C - s^*(C))\frac{\alpha}{\sqrt{C}}$. Full traversal across interval length $L=2$ requires $\Delta x_{\text{right}} \ge 2$, yielding the theoretical opening threshold:
    \begin{equation}
        \alpha^*(C) \approx \frac{2\sqrt{C}}{C - s^*(C)} \approx \frac{2}{\sqrt{C}} \quad (\text{for } C \gg s^*).
        \label{eq:alpha_star_derived}
    \end{equation}
    
    \item \textbf{Derivation of Sub-Threshold Pinned Oscillation ($\alpha < \alpha^*$):}
    When $\alpha < \alpha^*(C)$, rightward displacement $\Delta x_{\text{right}} < 2$ cannot cross the domain. Pinned against $+1$ during the non-burst phase, the trajectory sweeps linearly across $[1 - \Delta x_{\text{right}},\, 1]$. The cycle-averaged position is:
    \begin{equation}
        \bar{x} = 1 - \frac{\Delta x_{\text{right}}}{2} = 1 - \frac{\alpha(C - s^*(C))}{2\sqrt{C}}.
        \label{eq:pinned_formula}
    \end{equation}
    For $C=100$, Eq.~\eqref{eq:pinned_formula} predicts $\bar{x} = 0.620$ at $\alpha=0.10$ (observed: $0.602$) and $\bar{x} = 0.240$ at $\alpha=0.20$ (observed: $0.205$).
\end{enumerate}

\paragraph{Survival Window \& Revised $\alpha$-Scaling Law.}
The survival window ($\bar{x} < 0$) opens at $\alpha^*(C)$ and closes at an upper edge governed by clipping asymmetry ($\approx 1.8\alpha^*$). At large $C$, the revised scaling law $\alpha^*(C) \approx 2/\sqrt{C}$ implies that the upper survival edge drops as $\approx 3.6/\sqrt{C}$. For $C \ge 75$, $3.6/\sqrt{C} \le 0.41 < 0.80$, so the survival window drops entirely below $\alpha=0.80$, explaining the observed boundary disappearance.

\subsection{Small-$C$ Stochastic Arrival Timing Diagnostic}
At $C=10$, stochastic outcomes are governed by the arrival timing of late bursts before horizon $T$. Diagnostic evaluation ($N=500$ seeds, source: \nolinkurl{results/mechanism_probes/terminal_burst_timing_diagnostic.csv}) reveals a strong point-biserial correlation between failure ($x_T > 0.5$) and time elapsed since the last burst ($r_{\text{pb}} = 0.312$, $p = 9.8 \times 10^{-13}$). Seeds that fail experience an average gap of $11.76 \pm 0.41$ steps since their last burst, compared to $6.00 \pm 0.27$ steps for surviving seeds, confirming that small-$C$ terminal states reflect diffusion and burst arrival phase.
